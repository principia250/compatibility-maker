```typescript
'use client';

import { createContext, useContext, useState, useEffect } from 'react';

type Locale = 'en' | 'ja';

interface LocaleContextType {
  locale: Locale;
  setLocale: (locale: Locale) => void;
  t: (key: string) => string;
}

const LocaleContext = createContext<LocaleContextType | undefined>(undefined);

export function LanguageProvider({ children }: { children: React.ReactNode }) {
  const [locale, setLocale] = useState<Locale>('en');

  useEffect(() => {
    // ローカルストレージから言語設定を復元
    const savedLocale = localStorage.getItem('locale') as Locale;
    if (savedLocale && ['en', 'ja'].includes(savedLocale)) {
      setLocale(savedLocale);
    }
  }, []);

  useEffect(() => {
    // 言語設定をローカルストレージに保存
    localStorage.setItem('locale', locale);
    // HTMLのlang属性を更新
    document.documentElement.lang = locale;
  }, [locale]);

  const t = (key: string): string => {
    return translations[locale][key] || key;
  };

  return (
            <LanguageContext.Provider value={{ locale, setLocale, t }}>
          {children}
        </LanguageContext.Provider>
  );
}

export function useLanguage() {
  const context = useContext(LanguageContext);
  if (!context) {
    throw new Error('useLanguage must be used within a LanguageProvider');

  return context;
}
```

#### 3.3.1 CompatibilityMatrix コンポーネント

```typescript
// components/charts/CompatibilityMatrix.tsx
interface Element {
  id: string;
  name: string;
  side: 'left' | 'right';
  is_visible: boolean;
}

interface Compatibility {
  left_element_id: string;
  right_element_id: string;
  compatibility_score_id: string;
  notes?: string;
}

interface CompatibilityMatrixProps {
  elements: Element[];
  compatibilities: Compatibility[];
  onCompatibilityChange: (leftId: string, rightId: string, scoreId: string) => void;
  editable?: boolean;
}

export function CompatibilityMatrix({
  elements,
  compatibilities,
  onCompatibilityChange,
  editable = false
}: CompatibilityMatrixProps) {
  const leftElements = elements.filter(e => e.side === 'left' && e.is_visible);
  const rightElements = elements.filter(e => e.side === 'right' && e.is_visible);

  const getCompatibilityScore = (leftId: string, rightId: string) => {
    const compatibility = compatibilities.find(
      c => c.left_element_id === leftId && c.right_element_id === rightId
    );
    return compatibility?.compatibility_score_id || '';
  };

  const getScoreColor = (score: number) => {
    if (score === 2) return 'bg-green-500 text-white';
    if (score === 1) return 'bg-green-200 text-green-800';
    if (score === 0) return 'bg-gray-200 text-gray-800';
    if (score === -1) return 'bg-red-200 text-red-800';
    if (score === -2) return 'bg-red-500 text-white';
    return 'bg-gray-100 text-gray-600';
  };

  const { t } = useLanguage();

  const getScoreLabel = (score: number) => {
    if (score === 2) return t('score.very_strong');
    if (score === 1) return t('score.strong');
    if (score === 0) return t('score.even');
    if (score === -1) return t('score.weak');
    if (score === -2) return t('score.very_weak');
    return t('score.not_set');
  };

  return (
    <div className="overflow-x-auto">
      <div className="min-w-max">
        {/* ヘッダー行 */}
        <div className="flex">
          <div className="w-32 h-12 flex items-center justify-center font-medium text-gray-700 bg-gray-50 border-r border-b">
            要素
          </div>
          {rightElements.map(element => (
            <div key={element.id} className="w-24 h-12 flex items-center justify-center font-medium text-gray-700 bg-gray-50 border-b px-2 text-center">
              {element.name}
            </div>
          ))}
        </div>

        {/* データ行 */}
        {leftElements.map(leftElement => (
          <div key={leftElement.id} className="flex">
            <div className="w-32 h-12 flex items-center justify-center font-medium text-gray-700 bg-gray-50 border-r px-2 text-center">
              {leftElement.name}
            </div>
            {rightElements.map(rightElement => {
              const score = getCompatibilityScore(leftElement.id, rightElement.id);
              return (
                <div key={rightElement.id} className="w-24 h-12 border-r border-b flex items-center justify-center">
                  {editable ? (
                    <select
                      value={score}
                      onChange={(e) => onCompatibilityChange(
                        leftElement.id,
                        rightElement.id,
                        parseInt(e.target.value)
                      )}
                      className={`px-2 py-1 rounded text-sm font-medium ${getScoreColor(score)}`}
                    >
                      <option value={0}>未設定</option>
                      <option value={2}>大勝</option>
                      <option value={1}>勝ち</option>
                      <option value={0}>引き分け</option>
                      <option value={-1}>負け</option>
                      <option value={-2}>大負け</option>
                    </select>
                  ) : (
                    <div className={`px-2 py-1 rounded text-sm font-medium ${getScoreColor(score)}`}>
                      {getScoreLabel(score)}
                    </div>
                  )}
                </div>
              );
            })}
          </div>
        ))}
      </div>
    </div>
  );
}
```

#### 3.3.2 ElementList コンポーネント

```typescript
// components/charts/ElementList.tsx
interface ElementListProps {
  elements: Element[];
  side: 'left' | 'right';
  onElementAdd: (name: string, side: 'left' | 'right') => void;
  onElementUpdate: (id: string, updates: Partial<Element>) => void;
  onElementDelete: (id: string) => void;
  editable?: boolean;
}

export function ElementList({
  elements,
  side,
  onElementAdd,
  onElementUpdate,
  onElementDelete,
  editable = false
}: ElementListProps) {
  const [newElementName, setNewElementName] = useState('');
  const [editingId, setEditingId] = useState<string | null>(null);
  const [editingName, setEditingName] = useState('');

  const sideElements = elements.filter(e => e.side === side);

  const handleAddElement = () => {
    if (newElementName.trim()) {
      onElementAdd(newElementName.trim(), side);
      setNewElementName('');
    }
  };

  const handleStartEdit = (element: Element) => {
    setEditingId(element.id);
    setEditingName(element.name);
  };

  const handleSaveEdit = () => {
    if (editingId && editingName.trim()) {
      onElementUpdate(editingId, { name: editingName.trim() });
      setEditingId(null);
      setEditingName('');
    }
  };

  const handleCancelEdit = () => {
    setEditingId(null);
    setEditingName('');
  };

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
                 <h3 className="text-lg font-medium text-gray-900">
           {side === 'left' ? '左側要素' : '右側要素'}
         </h3>
        {editable && (
          <div className="flex space-x-2">
            <input
              type="text"
              value={newElementName}
              onChange={(e) => setNewElementName(e.target.value)}
              placeholder="新しい要素名"
              className="px-3 py-1 border border-gray-300 rounded-md text-sm"
              onKeyPress={(e) => e.key === 'Enter' && handleAddElement()}
            />
            <Button size="sm" onClick={handleAddElement}>
              追加
            </Button>
          </div>
        )}
      </div>

             <div className="space-y-2">
         {sideElements.map((element) => (
           <div key={element.id} className="flex items-center space-x-2 p-3 bg-white border border-gray-200 rounded-lg">
             <div className="flex-shrink-0 w-8 h-8 bg-gray-100 rounded-full flex items-center justify-center text-sm font-medium text-gray-600">
               {element.name.charAt(0)}
             </div>

            {editingId === element.id ? (
              <div className="flex-1 flex space-x-2">
                <input
                  type="text"
                  value={editingName}
                  onChange={(e) => setEditingName(e.target.value)}
                  className="flex-1 px-2 py-1 border border-gray-300 rounded text-sm"
                  onKeyPress={(e) => e.key === 'Enter' && handleSaveEdit()}
                />
                <Button size="sm" variant="primary" onClick={handleSaveEdit}>
                  保存
                </Button>
                <Button size="sm" variant="secondary" onClick={handleCancelEdit}>
                  キャンセル
                </Button>
              </div>
            ) : (
              <div className="flex-1">
                <span className="text-sm font-medium text-gray-900">
                  {element.name}
                </span>
              </div>
            )}

            {editable && (
              <div className="flex space-x-1">
                <Button
                  size="sm"
                  variant="ghost"
                  onClick={() => handleStartEdit(element)}
                >
                  編集
                </Button>
                <Button
                  size="sm"
                  variant="danger"
                  onClick={() => onElementDelete(element.id)}
                >
                  削除
                </Button>
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
```

```typescript
}

export const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // 初期認証状態の確認
    checkAuth();
  }, []);

  const checkAuth = async () => {
    try {
      const { data: { session } } = await supabase.auth.getSession();
      if (session) {
        await fetchUserProfile(session.user.id);
      }
    } catch (error) {
      console.error('認証チェックエラー:', error);
    } finally {
      setLoading(false);
    }
  };

  const fetchUserProfile = async (userId: string) => {
    const { data: profile } = await supabase
      .from('users')
      .select('*')
      .eq('id', userId)
      .single();

    if (profile) {
      setUser(profile);
    }
  };

  const login = async (email: string, password: string) => {
    const { data, error } = await supabase.auth.signInWithPassword({
      email,
      password
    });

    if (error) throw error;

    if (data.user) {
      await fetchUserProfile(data.user.id);
    }
  };

  const register = async (username: string, email: string, password: string) => {
    const { data, error } = await supabase.auth.signUp({
      email,
      password,
      options: { data: { username } }
    });

    if (error) throw error;

    if (data.user) {
      await fetchUserProfile(data.user.id);
    }
  };

  const logout = async () => {
    const { error } = await supabase.auth.signOut();
    if (error) throw error;

    setUser(null);
  };

  return (
    <AuthContext.Provider value={{
      user,
      loading,
      login,
      register,
      logout
    }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (context === undefined) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
}
```

### 5.2 相性図コンテキスト

```typescript
// contexts/ChartContext.tsx
interface ChartContextType {
  currentChart: Chart | null;
  elements: Element[];
  compatibilities: Compatibility[];
  updateElement: (id: string, updates: Partial<Element>) => void;
  addElement: (element: Omit<Element, 'id'>) => void;
  deleteElement: (id: string) => void;
  updateCompatibility: (leftId: string, rightId: string, score: number) => void;
  saveChart: () => Promise<void>;
}

export const ChartContext = createContext<ChartContextType | undefined>(undefined);

export function ChartProvider({
  children,
  chartId
}: {
  children: React.ReactNode;
  chartId: string;
}) {
  const [currentChart, setCurrentChart] = useState<Chart | null>(null);
  const [elements, setElements] = useState<Element[]>([]);
  const [compatibilities, setCompatibilities] = useState<Compatibility[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (chartId) {
      fetchChartData();
    }
  }, [chartId]);

  const fetchChartData = async () => {
    try {
      setLoading(true);

      // 相性図データ取得
      const [chartRes, elementsRes, compatibilitiesRes] = await Promise.all([
        fetch(`/api/charts/${chartId}`),
        fetch(`/api/charts/${chartId}/elements`),
        fetch(`/api/charts/${chartId}/compatibilities`)
      ]);

      const chart = await chartRes.json();
      const elementsData = await elementsRes.json();
      const compatibilitiesData = await compatibilitiesRes.json();

      if (chart.success) setCurrentChart(chart.data.chart);
      if (elementsData.success) setElements(elementsData.data.elements);
      if (compatibilitiesData.success) setCompatibilities(compatibilitiesData.data.compatibilities);

    } catch (error) {
      console.error('データ取得エラー:', error);
    } finally {
      setLoading(false);
    }
  };

  const updateElement = (id: string, updates: Partial<Element>) => {
    setElements(prev => prev.map(el =>
      el.id === id ? { ...el, ...updates } : el
    ));
  };

  const addElement = (element: Omit<Element, 'id'>) => {
    const newElement = {
      ...element,
      id: `temp-${Date.now()}` // 一時ID
    };
    setElements(prev => [...prev, newElement]);
  };

  const deleteElement = (id: string) => {
    setElements(prev => prev.filter(el => el.id !== id));
    // 関連する相性データも削除
    setCompatibilities(prev => prev.filter(c =>
      c.left_element_id !== id && c.right_element_id !== id
    ));
  };

  const updateCompatibility = (leftId: string, rightId: string, scoreId: string) => {
    setCompatibilities(prev => {
      const existing = prev.find(c =>
        c.left_element_id === leftId && c.right_element_id === rightId
      );

      if (existing) {
        return prev.map(c =>
          c.id === existing.id ? { ...c, compatibility_score_id: scoreId } : c
        );
      } else {
        return [...prev, {
          id: `temp-${Date.now()}`,
          left_element_id: leftId,
          right_element_id: rightId,
          compatibility_score_id: scoreId,
          created_at: new Date().toISOString()
        }];
      }
    });
  };

  const saveChart = async () => {
    try {
             // 要素の保存
       for (const element of elements) {
         if (element.id.startsWith('temp-')) {
           // 新規要素
           await fetch(`/api/charts/${chartId}/elements`, {
             method: 'POST',
             headers: { 'Content-Type': 'application/json' },
             body: JSON.stringify({
               name: element.name,
               side: element.side
             })
           });
         } else {
           // 既存要素の更新
           await fetch(`/api/charts/${chartId}/elements/${element.id}`, {
             method: 'PUT',
             headers: { 'Content-Type': 'application/json' },
             body: JSON.stringify({
               name: element.name,
               is_visible: element.is_visible
             })
           });
         }
       }

      // 相性データの保存
      for (const compatibility of compatibilities) {
        if (!compatibility.id.startsWith('temp-')) continue;

        await fetch(`/api/charts/${chartId}/compatibilities`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            left_element_id: compatibility.left_element_id,
            right_element_id: compatibility.right_element_id,
            compatibility_score_id: compatibility.compatibility_score_id
          })
        });
      }

      // データ再取得
      await fetchChartData();

    } catch (error) {
      console.error('保存エラー:', error);
      throw error;
    }
  };

  if (loading) {
    return <div className="flex justify-center items-center h-64">読み込み中...</div>;
  }

  return (
    <ChartContext.Provider value={{
      currentChart,
      elements,
      compatibilities,
      updateElement,
      addElement,
      deleteElement,
      updateCompatibility,
      saveChart
    }}>
      {children}
    </ChartContext.Provider>
  );
}

export function useChart() {
  const context = useContext(ChartContext);
  if (context === undefined) {
    throw new Error('useChart must be used within a ChartProvider');
  }
  return context;
}
```
